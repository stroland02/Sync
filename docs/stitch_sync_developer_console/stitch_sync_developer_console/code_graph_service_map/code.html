import requests
from core.circuit_breaker import Breaker

@Breaker(timeout=2000, fallback="queue")
class SalesforceGateway:
    def __init__(self, config):
        self.endpoint = config.get('SF_ENDPOINT')
        # Current timeout set to 5s. 
        # Consider lowering to fail fast?
        self.timeout = 5.0
        
    def sync_record(self, payload):
        res = requests.post(
            self.endpoint, 
            json=payload,
            timeout=self.timeout
        )
        return res.json()
